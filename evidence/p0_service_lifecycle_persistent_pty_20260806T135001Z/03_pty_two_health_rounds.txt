p0_check_services() {
>   local round="$1"
>   local failed=0
>   local pid_file="$PWD/research-runtime/service_pids"
> <er_simulator.server:app" "system_agent.server:app")
>   local expected_ports=(6102 6101 6100)
> <ime=%s\n' "$round" "$(date -u +%Y-%m-%dT%H:%M:%SZ)"
> <'PID_FILE_MISSING=%s\n' "$pid_file"; failed=1; else
>     mapfile -t p0_pids < "$pid_file"
>     printf 'PID_COUNT=%s PID_FILE=%s\n' "${#p0_pids[@]}" "$pid_file"
>     [[ "${#p0_pids[@]}" -eq 3 ]] || failed=1
>     local idx=0
>     for pid in "${p0_pids[@]}"; do
> < exe="" expected_module="${expected_modules[$idx]}"
>       stat=$(ps -o stat= -p "$pid" 2>/dev/null | xargs)
>       cmdline=$(tr '\0' ' ' < "/proc/$pid/cmdline" 2>/dev/null || true)
>       cwd=$(readlink "/proc/$pid/cwd" 2>/dev/null || true)
>       exe=$(readlink "/proc/$pid/exe" 2>/dev/null || true)
> <s CMD=%s\n' "$pid" "$stat" "$cwd" "$exe" "$cmdline"
>       kill -0 "$pid" 2>/dev/null || failed=1
>       [[ -n "$stat" && "$stat" != *Z* ]] || failed=1
>       [[ "$cwd" == "$PWD" ]] || failed=1
> <on"*"-m uvicorn"*"$expected_module"* ]] || failed=1
>       idx=$((idx+1))
>     done
>   fi
>   for port in 6100 6101 6102; do
>     local listener
>     listener=$(ss -ltnp 2>/dev/null | grep -E ":$port[[:space:]]" || true)
>     printf 'LISTENER_%s=%s\n' "$port" "$listener"
>     [[ -n "$listener" ]] || failed=1
>     local response curl_rc
> <%{http_code}' "http://127.0.0.1:$port/health" 2>&1)
>     curl_rc=$?
>     printf 'HEALTH_%s_RC=%s RESPONSE=%s\n' "$port" "$curl_rc" "$response"
> <& "$response" == *"HTTP_STATUS=200"* ]] || failed=1
>   done
>   local db_health db_pg db_pg_rc
> <mat '{{.State.Health.Status}}' 2>/dev/null || true)
>   db_pg=$(pg_isready -h 127.0.0.1 -p 6433 -U root 2>&1); db_pg_rc=$?
> <RC=%s DB_PG=%s\n' "$db_health" "$db_pg_rc" "$db_pg"
> <$db_pg" == *"accepting connections"* ]] || failed=1
>   printf '__P0_CHECK_END__ round=%s rc=%s\n' "$round" "$failed"
>   return "$failed"
> }
